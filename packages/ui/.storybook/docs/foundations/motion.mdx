export const literal = (reference) => {
  const value = primitive.primitive[reference.slice("{primitive.".length, -1)].$value;
  if (Array.isArray(value)) return `cubic-bezier(${value.join(", ")})`;
  return typeof value === "object" ? `${value.value}${value.unit}` : value;
};

export const motion = (property) =>
  Object.entries(semantic.semantic)
    .filter(([name]) => name.startsWith(`motion-${property}`))
    .map(([name, token]) => {
      const reduced =
        token.$extensions?.["com.bereghici.media"]?.["(prefers-reduced-motion: reduce)"];
      return {
        name: `ab-${name}`,
        value: literal(token.$value),
        usage: reduced === undefined ? "" : `${literal(reduced)} under reduced motion`,
      };
    });

# Motion

Motion is fast, eased out, interruptible and optional. It confirms what the user did and shows where things come from; it never decorates.

- **Animate `transform`, `opacity` and `filter` only.** Never `transition: all`.
- **Ease out to enter and exit,** never ease in. Exits are faster than entrances.
- **Use transitions, not keyframes,** for anything that can be triggered again before it ends, so it reverses from where it is.
- **Don't animate keyboard-initiated or very frequent actions** (more than about 100 a day).
- **Grow from the trigger,** at 0.97 scale, never from 0. Modals stay centred and enter at 0.95.
- **Hover motion is gated** behind `(hover: hover) and (pointer: fine)`, so touch never gets a stuck hover.
- **Springs** (`linear()` approximations, no script) are only for what the user moves directly: switch thumbs, segmented and tab indicators, the radio diamond, a toast's entrance. Everything the system starts uses ease-out.

## Reduced motion

Under `prefers-reduced-motion: reduce` the tokens remove movement: every scale, distance and blur becomes neutral and stagger becomes 0, while opacity and color fades stay. Components read the tokens, so apps get this with no code. Anything an app animates itself should use the same tokens, or check the media query.

## Durations and delays

Interface motion stays at or under 300ms. The 700ms entrance plays once per visit, and the toast durations are how long a toast stays, not an animation.
