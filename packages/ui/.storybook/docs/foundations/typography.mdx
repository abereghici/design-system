export const family = (key) => {
  const styles = tokens.type.groups
    .filter((group) => group.family === key)
    .flatMap((group) => group.styles);
  return {
    fontFamily: tokens.type.families[key],
    fontSizes: [...new Set(styles.map((style) => style.fontSize))],
    fontWeight: styles[0].fontWeight,
    sampleText: "Sphinx of black quartz, judge my vow",
  };
};

# Typography

Noto Serif sets headings and display statements, Figtree sets everything people read and operate, and code uses the system monospace stack. Both families include a true italic. The UIProvider loads the fonts, so apps add nothing.

Set text with a type role: `font: var(--ab-type-body-md)` plus `letter-spacing: var(--ab-type-body-md-letter-spacing)`, because the `font` shorthand can't carry letter spacing. The roles keep size, weight and line height in step.

- Sizes are in `rem`, so text follows the reader's default size.
- Line heights are unitless.
- The larger steps grow a little with the window and still follow the reader's text size, as WCAG 1.4.4 asks.
